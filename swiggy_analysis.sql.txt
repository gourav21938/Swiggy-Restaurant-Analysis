-- Q1: Total restaurants on Swiggy
SELECT COUNT(*) AS total_restaurants 
FROM swiggy;

-- Q2: Top 10 cities by number of restaurants
SELECT city, COUNT(*) AS total
FROM swiggy
GROUP BY city
ORDER BY total DESC
LIMIT 10;

-- Q3: Top 10 cuisines across India
SELECT cuisine, COUNT(*) AS count
FROM swiggy
GROUP BY cuisine
ORDER BY count DESC
LIMIT 10;

-- Q4: Average cost per city (top 10 most expensive)
SELECT city,
       ROUND(AVG(CAST(REPLACE(REPLACE(cost, '₹ ', ''), ',', '') AS INTEGER)), 2) AS avg_cost
FROM swiggy
WHERE cost IS NOT NULL
GROUP BY city
ORDER BY avg_cost DESC
LIMIT 10;

-- Q5: Highest rated restaurants (ignore '--' and 'Too Few Ratings')
SELECT name, city, rating, rating_count, cost
FROM swiggy
WHERE rating != '--' AND rating IS NOT NULL
ORDER BY CAST(rating AS REAL) DESC
LIMIT 10;

-- Q6: Cities with best average rating
SELECT city,
       ROUND(AVG(CAST(rating AS REAL)), 2) AS avg_rating,
       COUNT(*) AS total_restaurants
FROM swiggy
WHERE rating != '--' AND rating IS NOT NULL
GROUP BY city
ORDER BY avg_rating DESC
LIMIT 10;

-- Q7: Most popular restaurant chains (in most cities)
SELECT name,
       COUNT(DISTINCT city) AS cities_present
FROM swiggy
GROUP BY name
ORDER BY cities_present DESC
LIMIT 10;

-- Q8: Budget restaurants with high ratings (cost under 300, rating above 4)
SELECT name, city, cuisine, cost, rating
FROM swiggy
WHERE CAST(REPLACE(REPLACE(cost, '₹ ', ''), ',', '') AS INTEGER) < 300
  AND rating != '--'
  AND CAST(rating AS REAL) > 4.0
ORDER BY CAST(rating AS REAL) DESC
LIMIT 15;

-- Q9: Most diverse cities by cuisine variety
SELECT city,
       COUNT(DISTINCT cuisine) AS cuisine_variety
FROM swiggy
GROUP BY city
ORDER BY cuisine_variety DESC
LIMIT 10;

-- Q10: Lowest rated restaurants with significant reviews
SELECT name, city, cuisine, rating, rating_count
FROM swiggy
WHERE rating != '--'
  AND rating_count NOT LIKE 'Too Few Ratings'
  AND CAST(rating AS REAL) < 3.5
ORDER BY CAST(rating AS REAL) ASC
LIMIT 10;
